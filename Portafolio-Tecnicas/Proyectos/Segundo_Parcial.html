#install.packages("igraph")
library(tidyverse)
library(igraph)
#Primero cargamos las librerias tidyverse para la maninpulacion de datos e igraph para construir y analizar la red
acciones <- read_csv("acionesInspeccionvsrm.csv")
#Cargamos la base de datos a nuestro proyecto
#Utilizamos la IA para filtrar los datos donde la variable de inspeccion sea mayor a 0
acciones_filtrado <- subset(acciones, inspeccion > 0)
#De esta manera ya tenemos nuestros datos filtrados y listos para usar
#Creamos una edgelist con los datos ya filtrados
edgelist <- acciones_filtrado %>%
select(estado, materia, inspeccion)

#En esta parte creamos una lista de enlaces en donde el estado es nuestro nodo tipo 1, la materia es nuestro nodo tipo 2 y la inspeccion es el peso (Weight).
#Utilizamos la libreria igraph para crear un grafico no dirigo
red <- graph_from_data_frame(edgelist, directed = FALSE)
#Asignamos el peso a los numeros de inspecciones
E(red)$weight <- edgelist$inspeccion
#Visualizamos la red con un plot, mejorando la distribucion para un mejor analisis
plot(red,
  layout = layout_with_fr,
  vertex.label = E(red)$weight,
  vertex.label.cex = 0.9,
  edge.arrow.size = 0,
main = "Red de Inspecciones ambientales por estado")
